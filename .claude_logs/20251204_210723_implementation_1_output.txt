### 📝 Usage Example

```go
// Initialize client
ctx := context.Background()
config := redis.DefaultConfig("redis://localhost:6379/0")
client, err := redis.NewClient(ctx, config)
defer client.Close()

// Rate limiting
count, _ := client.Incr(ctx, "ratelimit:ip:192.168.1.1")
if count == 1 {
    client.Expire(ctx, "ratelimit:ip:192.168.1.1", time.Minute)
}

// URL caching
client.Set(ctx, "url:aB3xY9", "https://example.com", time.Hour)
url, _ := client.Get(ctx, "url:aB3xY9")

// Health monitoring
monitor := redis.NewMonitor(client, 30*time.Second)
monitor.Start(ctx)
```
